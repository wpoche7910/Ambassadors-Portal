function csvCell(v) { v = v == null ? "" : String(v); if (/^[=+\-@\t\r]/.test(v)) v = "'" + v; return /[",\n\r]/.test(v) ? '"' + v.replace(/"/g, '""') + '"' : v; }

function download(name, rows) {

  var blob = new Blob(["﻿" + rows.map(function (r) { return r.map(csvCell).join(","); }).join("\r\n")], { type: "text/csv;charset=utf-8" });

var routing = 0;

async function route() {

  var my = ++routing, h = location.hash.replace(/^#/, "") || "/home", i = h.indexOf("?");

  var path = i < 0 ? h : h.slice(0, i), params = new URLSearchParams(i < 0 ? "" : h.slice(i + 1)), parts = path.split("/").filter(Boolean);

  try {

    if (path === "/join") return joinView(params.get("code"));

    if (recovering) return recoveryView();

    if (!session) return loginView();

    if (!staff || staff.user_id !== session.user.id) {

      var rows = await q(sb.from("staff").select("id,user_id,email,name,role,active,invite_expires,created_at").eq("user_id", session.user.id));

      staff = rows[0] && rows[0].active ? rows[0] : null;

    }

    if (!staff) return deniedView();

    if (my !== routing) return;

    var page = parts[0], arg = parts[1];

    if (page === "login") return go("#/home");

    var pages = { home: homePage, calendar: function () { return calendarPage(params); }, members: rosterPage, search: function () { return searchPage(params); }, categories: categoriesPage,

      export: exportPage, activity: activityPage, users: function () { return usersPage(); }, account: accountPage, event: function () { return arg ? eventPage(arg, params) : go("#/calendar"); }, member: function () { return arg ? memberPage(arg) : go("#/members"); } };

    await (pages[page] || homePage)(); window.scrollTo(0, 0);

  } catch (e) {

    console.error(e); if (my === routing) { plain('<div class="card"><h2>Something went wrong</h2><p>' + esc(e.message || e) + '</p><p><a href="#/home">Back to home</a> · <button class="linklike" onclick="location.reload()">Try again</button></p></div>'); }

  }

}
